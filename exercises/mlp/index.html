    """Exercises 2–4 — MLP from scratch (ReLU hidden, SGD).

sklearn entra só para gerar/splitar os dados, a regressão logística
de baseline e a matriz de confusão. Forward, loss, gradiente e update
são os mesmos métodos em todos os exercícios.

Uso (na raiz do repositório):

    python docs/exercises/mlp/code/mlp.py
"""

from __future__ import annotations

from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np
from sklearn.datasets import make_blobs, make_classification
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import confusion_matrix
from sklearn.model_selection import train_test_split

FIGURES = Path(__file__).resolve().parents[1] / "figures"
RNG = np.random.default_rng(42)


class MLP:
    """MLP de profundidade arbitrária. ``sizes`` é a lista [d_in, h1, ..., d_out].

    Camadas ocultas: ReLU. Saída: sigmoide (binário) ou softmax (multiclasse).
    """

    def __init__(self, sizes: list[int], rng: np.random.Generator, output: str):
        self.sizes = list(sizes)
        self.output = output
        self.W, self.b = [], []
        for n_in, n_out in zip(self.sizes[:-1], self.sizes[1:]):
            scale = np.sqrt(2.0 / n_in)
            self.W.append(rng.normal(0.0, scale, size=(n_in, n_out)))
            self.b.append(np.zeros(n_out))

    def forward(self, X: np.ndarray) -> np.ndarray:
        self.cache = {"a0": X}
        a = X
        last = len(self.W) - 1
        for i, (W, b) in enumerate(zip(self.W, self.b)):
            z = a @ W + b
            self.cache[f"z{i+1}"] = z
            if i == last:
                a = _sigmoid(z) if self.output == "binary" else _softmax(z)
            else:
                a = np.maximum(0.0, z)
            self.cache[f"a{i+1}"] = a
        return a

    def loss(self, yhat: np.ndarray, y: np.ndarray) -> float:
        eps = 1e-12
        if self.output == "binary":
            y = y.reshape(-1, 1)
            return float(-np.mean(y * np.log(yhat + eps) + (1 - y) * np.log(1 - yhat + eps)))
        return float(-np.mean(np.log(yhat[np.arange(len(y)), y] + eps)))

    def backward(self, y: np.ndarray) -> tuple[list[np.ndarray], list[np.ndarray]]:
        # BCE+sigmoide e CCE+softmax compartilham dL/du = (ŷ − y) / N.
        n = self.cache["a0"].shape[0]
        L = len(self.W)
        yhat = self.cache[f"a{L}"]
        if self.output == "binary":
            dz = (yhat - y.reshape(-1, 1)) / n
        else:
            y_oh = np.zeros_like(yhat)
            y_oh[np.arange(n), y] = 1.0
            dz = (yhat - y_oh) / n

        dW, db = [None] * L, [None] * L
        for i in reversed(range(L)):
            a_prev = self.cache[f"a{i}"]
            dW[i] = a_prev.T @ dz
            db[i] = dz.sum(axis=0)
            if i > 0:
                da = dz @ self.W[i].T
                dz = da * (self.cache[f"z{i}"] > 0)

        return dW, db

    def update(self, dW: list[np.ndarray], db: list[np.ndarray], lr: float) -> None:
        for i in range(len(self.W)):
            self.W[i] -= lr * dW[i]
            self.b[i] -= lr * db[i]

    def predict(self, X: np.ndarray) -> np.ndarray:
        yhat = self.forward(X)
        if self.output == "binary":
            return (yhat.ravel() >= 0.5).astype(int)
        return yhat.argmax(axis=1)

    def n_params(self) -> int:
        return int(sum(W.size + b.size for W, b in zip(self.W, self.b)))


def _sigmoid(z: np.ndarray) -> np.ndarray:
    z = np.clip(z, -50.0, 50.0)
    return 1.0 / (1.0 + np.exp(-z))


def _softmax(z: np.ndarray) -> np.ndarray:
    z = z - z.max(axis=1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=1, keepdims=True)


def train(
    model: MLP,
    X: np.ndarray,
    y: np.ndarray,
    lr: float,
    epochs: int,
    batch_size: int,
    rng: np.random.Generator,
) -> list[float]:
    losses: list[float] = []
    n = len(X)
    for _ in range(epochs):
        order = rng.permutation(n)
        Xs, ys = X[order], y[order]
        for start in range(0, n, batch_size):
            xb = Xs[start : start + batch_size]
            yb = ys[start : start + batch_size]
            model.forward(xb)
            dW, db = model.backward(yb)
            model.update(dW, db, lr)
        yhat = model.forward(X)
        losses.append(model.loss(yhat, y))
    return losses


def accuracy(model: MLP, X: np.ndarray, y: np.ndarray) -> float:
    return float(np.mean(model.predict(X) == y))


def save(fig, name: str) -> None:
    fig.tight_layout()
    fig.savefig(FIGURES / name, dpi=150)
    plt.close(fig)


def plot_loss(ax, losses, title: str, color="#4C78A8") -> None:
    ax.plot(range(1, len(losses) + 1), losses, color=color)
    ax.set_xlabel("epoch")
    ax.set_ylabel("loss (treino)")
    ax.set_title(title)
    ax.grid(True, alpha=0.3)

    Exercise 1

    One Step of Backpropagation, by Hand

    MLP \(2 \to 2 \to 1\), \(\tanh\) nas duas camadas, \(L=(y-\hat y)^2\), \(\mathbf{x}=[0{,}5,\,-0{,}2]^\top\), \(y=1\), \(\eta=0{,}3\). Convenção: \(\mathbf{z}^{(1)}=\mathbf{W}^{(1)}\mathbf{x}+\mathbf{b}^{(1)}\), linhas de \(\mathbf{W}^{(1)}\) = neurônios. Números com 4 casas (e um dígito extra nos intermediários).

    A — Forward pass

    \[ \mathbf{z}^{(1)} = \begin{bmatrix}0.3 & -0.1\\ 0.2 & 0.4\end{bmatrix} \begin{bmatrix}0.5\\ -0.2\end{bmatrix} + \begin{bmatrix}0.1\\ -0.2\end{bmatrix} = \begin{bmatrix}0.15+0.02\\ 0.10-0.08\end{bmatrix} + \begin{bmatrix}0.1\\ -0.2\end{bmatrix} = \begin{bmatrix}0.2700\\ -0.1800\end{bmatrix}. \]
    \[ \mathbf{h}^{(1)}=\tanh(\mathbf{z}^{(1)}) = \begin{bmatrix}\tanh(0.2700)\\ \tanh(-0.1800)\end{bmatrix} = \begin{bmatrix}0.2636\\ -0.1781\end{bmatrix}. \]
    \[ u^{(2)} = [0.5,\,-0.3]\,\mathbf{h}^{(1)}+0.2 = 0.5\cdot 0.2636 + (-0.3)\cdot(-0.1781)+0.2 = 0.1318+0.0534+0.2 = 0.3852. \]
    \[ \hat y=\tanh(0.3852)=0.3672, \qquad L=(1-0.3672)^2=0.4004. \]

    B — Backward pass

    \(L=(y-\hat y)^2\) \(\Rightarrow\) \(\partial L/\partial\hat y=-2(y-\hat y)=-2(0.6328)=-1.2655\).

    Derivada do \(\tanh\): \(1-\tanh^2(u^{(2)})=1-\hat y^2=1-0.1349=0.8651\). Cadeia:

    \[ \frac{\partial L}{\partial u^{(2)}} = \frac{\partial L}{\partial\hat y}\,(1-\hat y^2) = (-1.2655)\cdot 0.8651 = -1.0948. \]

    Camada de saída (\(\mathbf{W}^{(2)}\) é um vetor linha):

    \[ \frac{\partial L}{\partial\mathbf{W}^{(2)}} = \frac{\partial L}{\partial u^{(2)}}\,\mathbf{h}^{(1)} = -1.0948\cdot \begin{bmatrix}0.2636\\ -0.1781\end{bmatrix} = \begin{bmatrix}-0.2886\\ 0.1950\end{bmatrix}^\top, \qquad \frac{\partial L}{\partial b^{(2)}} = -1.0948. \]

    Propagação para a oculta:

    \[ \frac{\partial L}{\partial\mathbf{h}^{(1)}} = \frac{\partial L}{\partial u^{(2)}}\,\mathbf{W}^{(2)} = -1.0948\cdot[0.5,\,-0.3] = [-0.5474,\,0.3284]. \]
    \[ \frac{\partial L}{\partial\mathbf{z}^{(1)}} = \frac{\partial L}{\partial\mathbf{h}^{(1)}}\odot\bigl(1-(\mathbf{h}^{(1)})^2\bigr). \]

    \(1-h_1^2=1-0.0695=0.9305\), \(1-h_2^2=1-0.0317=0.9683\), logo

    \[ \frac{\partial L}{\partial\mathbf{z}^{(1)}} = [-0.5474\cdot 0.9305,\; 0.3284\cdot 0.9683] = [-0.5094,\,0.3180]. \]
    \[ \frac{\partial L}{\partial\mathbf{W}^{(1)}} = \frac{\partial L}{\partial\mathbf{z}^{(1)}}\,\mathbf{x}^\top = \begin{bmatrix}-0.5094\\ 0.3180\end{bmatrix} [0.5,\,-0.2] = \begin{bmatrix} -0.2547 & 0.1019\\ 0.1590 & -0.0636 \end{bmatrix}, \qquad \frac{\partial L}{\partial\mathbf{b}^{(1)}} = [-0.5094,\,0.3180]. \]

    C — Parameter update

    \(\theta\leftarrow\theta-\eta\,\partial L/\partial\theta\) com \(\eta=0.3\):

    \[ \mathbf{W}^{(2)} = [0.5,\,-0.3]-0.3[-0.2886,\,0.1950] = [0.5866,\,-0.3585], \qquad b^{(2)}=0.2-0.3(-1.0948)=0.5284. \]
    \[ \mathbf{W}^{(1)} = \begin{bmatrix}0.3 & -0.1\\ 0.2 & 0.4\end{bmatrix} - 0.3 \begin{bmatrix}-0.2547 & 0.1019\\ 0.1590 & -0.0636\end{bmatrix} = \begin{bmatrix}0.3764 & -0.1306\\ 0.1523 & 0.4191\end{bmatrix}, \]
    \[ \mathbf{b}^{(1)} = [0.1,\,-0.2]-0.3[-0.5094,\,0.3180] = [0.2528,\,-0.2954]. \]

    Um segundo forward com esses parâmetros dá \(L'=0.0834<0.4004\). A loss desce: o passo foi exatamente \(-\eta\nabla L\), portanto (para \(\eta\) pequeno o bastante) \(L\) diminui na direção de descida.

    Exercise 2

    Binary Classification a Straight Line Cannot Do

    A — Generate the data

    make_blobs com os três centros do enunciado, y=(c>0) — a classe 1 fica nos dois extremos da diagonal, a classe 0 no meio. Split 80/20 estratificado, random_state=42.

    B — Establish the baseline

    LogisticRegression no treino, avaliada no teste: 47,50%. Abaixo de 50%, como o enunciado antecipa (~48%).

    C — Implement the MLP

    Arquitetura [2, 16, 1]: oculta ReLU, saída sigmoide, loss BCE. SGD em minibatches de 64, \(\eta=0.08\), 800 epochs. Acurácia no teste: 92,50% (faixa 90–93%).

    D — Analysis

    O baseline linear está abaixo de 50% porque a geometria é a de um XOR contínuo: a classe 1 não é um lado da reta, são dois lados. Uma reta que classifica o canto superior-direito como 1 empurra o inferior-esquerdo para 0 — e vice-versa. No meio ainda mora a classe 0, então qualquer corte único acerta um cluster laranja e erra o outro mais os azuis da fronteira. Resultado: pior que o chute.

    A Figura 3 mostra o que a camada oculta compra: a região da classe 1 é desconexa (duas componentes). Isso não é um semiplano \(\mathbf{w}^\top\mathbf{x}+b>0\). Os 16 ReLUs recortam o plano em polígonos e a saída combina dois deles; daí os 92,5%.

    Exercise 3

    Multi-Class, Same Network

    A — Generate the data

    make_classification: 1500 amostras, 4 features informativas, 3 classes, 2 clusters por classe, class_sep=1.0, random_state=42. Split 80/20 estratificado com a mesma semente.

    B — Adapt the network

    Mesmo forward / backward / update. Só mudam sizes=[4, 32, 3] e output="multiclass" (softmax + CCE).

    Derivação de \(\partial L/\partial\mathbf{u}\). Para uma amostra, \(p_k=e^{u_k}/\sum_j e^{u_j}\) e \(L=-\sum_k y_k\log p_k\) com \(\sum_k y_k=1\), \(y_k\in\{0,1\}\).

    \[ \log p_k = u_k - \log\sum_j e^{u_j}, \qquad \frac{\partial\log p_k}{\partial u_i} = \mathbf{1}_{k=i}-p_i. \]
    \[ \frac{\partial L}{\partial u_i} = -\sum_k y_k(\mathbf{1}_{k=i}-p_i) = -y_i + p_i\sum_k y_k = p_i-y_i. \]

    No batch, a média dá \((\hat{\mathbf{y}}-\mathbf{y}_{\text{oh}})/N\) — a mesma forma que BCE+sigmoide no caso binário, por isso o backward não foi reescrito.

    C — Train and evaluate

    [4, 32, 3], 259 parâmetros, \(\eta=0.08\), 400 epochs, batch 64. Acurácia no teste: 85,67% (faixa 83–86%). Baseline linear no mesmo split: 66,33% (~66%).

    D — Analysis

    A maior célula fora da diagonal vale 9, empatada em \((0\to 2)\), \((1\to 0)\) e \((1\to 2)\). Somando os dois sentidos, os pares 0–1 e 1–2 empatam com 16 erros; 0–2 tem 11. O par mais confundido, portanto, envolve a classe 1 com cada vizinha.

    O gerador cria 6 clusters em 4D (2 por classe). Com class_sep=1.0 esses clusters se encostam: um dos dois modos da classe 1 pode cair perto de um modo da 0, o outro perto da 2 — daí a classe 1 aparecer nos dois pares. A camada oculta sobe o baseline de 66,3% para 85,7%: a logística só corta o \(\mathbb{R}^4\) em 3 semi-espaços; o ReLU 32-D separa modos da mesma classe que não são convexos juntos.

    Discussão

    O ponto fácil de errar no Exercise 1 é a orientação de \(\mathbf{W}^{(1)}\): linhas = neurônios, \(\partial L/\partial\mathbf{W}^{(1)}=(\partial L/\partial\mathbf{z})\mathbf{x}^\top\). No código, o atalho \(\partial L/\partial\mathbf{u}=\hat y-y\) vale para os dois pares (sigmoide+BCE e softmax+CCE) — por isso o mesmo backward serve nos quatro exercícios depois do 1.

    Conclusão

    A camada oculta compra regiões desconexas (Exercise 2) e modos não-convexos por classe (Exercise 3). Uma segunda oculta, no mesmo orçamento, não compra acurácia neste gerador: só loss de treino menor e mais variância entre sementes.
